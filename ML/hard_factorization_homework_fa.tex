\section*{سوال ۳: به پرسش‌های زیر پاسخ دهید}

\setcounter{enumi}{90}
\begin{enumerate}
\item اگر $\alpha$ و $\beta$ ریشه‌های $x^2 - 7x + 10 = 0$ باشند، مقدار $\alpha^2 + \beta^2$ را بیابید.
\item اگر $\alpha$ و $\beta$ ریشه‌های $2x^2 + 5x - 3 = 0$ باشند، مقدار $\dfrac{1}{\alpha} + \dfrac{1}{\beta}$ را بیابید.
\item اگر $\alpha$ و $\beta$ ریشه‌های $x^2 + 3x - 5 = 0$ باشند، مقدار $\alpha^3 + \beta^3$ را بیابید.
\item اگر $\alpha$ و $\beta$ ریشه‌های $x^2 - 6x + 4 = 0$ باشند، مقدار $(\alpha - \beta)^2$ را بیابید.
\item اگر $\alpha$ و $\beta$ ریشه‌های $3x^2 - 2x - 7 = 0$ باشند، مقدار $\alpha^2 \beta + \alpha \beta^2$ را بیابید.
\item اگر $\alpha$ و $\beta$ ریشه‌های $x^2 + px + q = 0$ باشند، $\alpha^5 + \beta^5$ را بر حسب $p$ و $q$ بیابید.
\item اگر $\alpha$ و $\beta$ ریشه‌های $x^2 + 7x + 1 = 0$ باشند، مقدار $\dfrac{\alpha}{\beta} + \dfrac{\beta}{\alpha}$ را بیابید.
\item اگر $\alpha$ و $\beta$ ریشه‌های $x^2 - x - 1 = 0$ باشند، مقدار $\alpha^3 + \beta^3$ را بیابید و نشان دهید برابر $2\alpha + 2\beta + 3$ است.
\item معادله‌ای با ضرایب صحیح بیابید که $3 + \sqrt{7}$ یکی از ریشه‌های آن باشد.
\item معادله‌ای با ضرایب صحیح بیابید که $\sqrt{5} - \sqrt{3}$ یکی از ریشه‌های آن باشد.
\item ریشه‌های $x^2 - 3x + k = 0$ برابر $\alpha$ و $\beta$ هستند. $k$ را طوری بیابید که $\alpha - \beta = 5$.
\item ریشه‌های $x^2 + px + 12 = 0$ اختلافشان ۱ است. همه مقادیر ممکن $p$ را بیابید.
\item $k$ را طوری بیابید که یکی از ریشه‌های $x^2 + (k+1)x + 2k = 0$ دو برابر ریشه دیگر باشد.
\item $m$ را طوری بیابید که ریشه‌های $x^2 - mx + m + 7 = 0$ برابر باشند.
\item معادله‌ای بیابید که ریشه‌های آن $\alpha + 1$ و $\beta + 1$ باشند، جایی که $\alpha, \beta$ ریشه‌های $x^2 - 4x + 2 = 0$ هستند.
\item معادله‌ای بیابید که ریشه‌های آن $\alpha^2 + \dfrac{1}{\beta^2}$ و $\beta^2 + \dfrac{1}{\alpha^2}$ باشند، جایی که $\alpha, \beta$ ریشه‌های $x^2 - 3x + 1 = 0$ هستند.
\item برای چه مقادیر $k$، معادله $x^2 + kx + 9 = 0$ ریشه‌های برابر دارد؟
\item برای چه مقادیر $k$، معادله $2x^2 + 3x + k = 0$ دو ریشه حقیقی متمایز دارد؟
\item برای چه مقادیر $m$، معادله $x^2 + (m+2)x + m + 5 = 0$ ریشه حقیقی ندارد؟
\item برای چه مقادیر $p$، معادله $(p+1)x^2 - 2(p-1)x + p - 2 = 0$ ریشه‌های برابر دارد؟
\item برای چه مقادیر حقیقی $k$، معادله $x^2 + (k+1)x + k^2 = 0$ حداقل یک ریشه حقیقی دارد؟
\item نشان دهید که $x^2 + 2ax + (a^2 + a + 1) = 0$ برای هر $a > -1$ ریشه حقیقی ندارد.
\item اگر $x^2 + px + q = 0$ و $x^2 + qx + p = 0$ ریشه مشترک داشته باشند، نشان دهید که $p + q + 1 = 0$ یا $p = q$.
\item اگر $x^2 + ax + b = 0$ و $x^2 + bx + a = 0$ ریشه مشترک داشته باشند، ریشه مشترک را بر حسب $a$ و $b$ بیان کنید.
\item همه مقادیر $k$ را بیابید که $x^2 - kx + 12 = 0$ و $x^2 + kx + 12 = 0$ ریشه مشترک دارند.
\item برای چه مقادیر $m$، هر دو ریشه $x^2 - 2mx + m^2 - 1 = 0$ بین $-2$ و $4$ قرار دارند؟
\item برای چه مقادیر $m$، معادله $x^2 + 2mx + m^2 - 4 = 0$ دارد:
  \begin{enumerate}[label=(\alph*)]
    \item هر دو ریشه مثبت
    \item هر دو ریشه منفی
    \item ریشه‌های علامت مخالف
  \end{enumerate}
\item $k$ را طوری بیابید که $x^2 - 2kx + k^2 - 5 = 0$ داشته باشد:
  \begin{enumerate}[label=(\alph*)]
    \item هر دو ریشه بزرگتر از ۱
    \item هر دو ریشه کوچکتر از ۳
    \item عدد ۲ بین ریشه‌ها قرار گیرد
  \end{enumerate}
\item برای چه مقادیر $a$، معادله $x^2 - 2ax + a^2 - 9 = 0$ دقیقاً یک ریشه در بازه $(0, 5)$ دارد؟
\item برای چه مقادیر $m$، هر دو ریشه $x^2 + mx + 4 = 0$ حقیقی و در بازه $(-1, 3)$ هستند؟
\item برای چه مقادیر $k$، معادله $x^2 + kx + 3 = 0$ یک ریشه در $(0, 1)$ و ریشه دیگر در $(2, 3)$ دارد؟
\item $k$ را طوری بیابید که مجموع ریشه‌های $x^2 + (2k-1)x + k^2 - 3 = 0$ برابر حاصل‌ضرب ریشه‌ها باشد.
\item $p$ را طوری بیابید که یکی از ریشه‌های $x^2 + px + 12 = 0$ مربع ریشه دیگر باشد.
\item $k$ را طوری بیابید که ریشه‌های $x^2 + (3k-1)x + 2k^2 + 2k - 4 = 0$ اختلافشان ۲ باشد.
\item اگر $\alpha, \beta$ ریشه‌های $x^2 + mx + m^2 - 7 = 0$ باشند، $m$ را طوری بیابید که $\alpha^2 + \beta^2 = 5$.
\item همه $k$‌های حقیقی را بیابید که ریشه‌های $\alpha, \beta$ معادله $x^2 - 2kx + k^2 - 4 = 0$ شرط $|\alpha| + |\beta| = 6$ را برآورده کنند.
\item اگر $\alpha, \beta$ ریشه‌های $x^2 + px + q = 0$ باشند، معادله‌ای بیابید که ریشه‌های آن $\dfrac{\alpha}{\alpha+1}$ و $\dfrac{\beta}{\beta+1}$ باشند.
\item اگر $\alpha, \beta$ ریشه‌های $x^2 + 5x + 3 = 0$ باشند، معادله‌ای بیابید که ریشه‌های آن $(\alpha+\beta)^2$ و $(\alpha-\beta)^2$ باشند.
\item اگر $\alpha, \beta$ ریشه‌های $x^2 - x - 4 = 0$ باشند، معادله‌ای بیابید که ریشه‌های آن $\dfrac{1}{\alpha+\beta}$ و $\dfrac{1}{\alpha\beta}$ باشند.
\item اگر $\alpha, \beta$ ریشه‌های $2x^2 + 3x + 5 = 0$ باشند، معادله‌ای بیابید که ریشه‌های آن $\alpha + \dfrac{1}{\beta}$ و $\beta + \dfrac{1}{\alpha}$ باشند.
\end{enumerate}

\section*{سوالات چالشی (اختیاری)}

\setcounter{enumi}{130}
\begin{enumerate}
\item فرض کنید $\alpha, \beta$ ریشه‌های $x^2 + x + 1 = 0$ باشند. مقدار $\alpha^{2025} + \beta^{2025}$ را بیابید.
\item فرض کنید $\alpha, \beta$ ریشه‌های $x^2 - 2017x + 1 = 0$ باشند. مقدار $\alpha^{2017} + \beta^{2017}$ را بر پیمانه ۱۰ بیابید.
\item اگر $\alpha, \beta$ ریشه‌های $x^2 + x - 1 = 0$ باشند، مقدار $\alpha^5 + \beta^5$ را بیابید.
\item فرض کنید $\alpha, \beta, \gamma$ ریشه‌های $x^3 - 3x^2 + 4x - 2 = 0$ باشند. $\alpha^2 + \beta^2 + \gamma^2$، $\alpha\beta + \beta\gamma + \gamma\alpha$ و $\alpha\beta\gamma$ را بیابید.
\item اگر $\alpha, \beta$ ریشه‌های $x^2 + x + 1 = 0$ باشند، $\alpha^{100} + \beta^{100}$ و $\alpha \cdot \beta^{100} + \beta \cdot \alpha^{100}$ را محاسبه کنید.
\item با دانستن اینکه $\alpha$ و $\beta$ ریشه‌های $x^2 + 7x + 2 = 0$ هستند، $\lfloor \alpha \rfloor + \lfloor \beta \rfloor$ را بیابید ($\lfloor \cdot \rfloor$ تابع کف است).
\item اگر $\alpha, \beta$ ریشه‌های $x^2 - 6x + 4 = 0$ باشند، مقدار $\sqrt{\alpha} + \sqrt{\beta}$ را بیابید.
\item فرض کنید $r_1, r_2$ ریشه‌های $x^2 + 2024x + 2025 = 0$ باشند. $(r_1 + 1)(r_2 + 1)$ را محاسبه کنید.
\item اگر $\alpha, \beta$ ریشه‌های $x^2 + x + 2 = 0$ باشند، $\alpha^8 + \beta^8$ را بیابید.
\item فرض کنید $\alpha$ ریشه $x^2 - x - 1 = 0$ باشد. $\alpha^4 - 3\alpha^2$ را بدون محاسبه صریح $\alpha$ بیابید.
\item فرض کنید $\alpha, \beta$ ریشه‌های $x^2 + 3x + 9 = 0$ باشند. $\alpha^3 + \beta^3$ و $\alpha^6 + \beta^6$ را بیابید.
\item اگر $\alpha$ و $\beta$ ریشه‌های $x^2 - x + 1 = 0$ باشند، $\alpha^{2026} + \beta^{2026}$ را بیابید.
\item اگر $r$ ریشه $x^3 - 2x^2 + x - 3 = 0$ باشد، مقدار $r^4 - 2r^3 + r^2$ را بیابید.
\item معادله $x^2 + bx + c = 0$ ریشه‌های $\alpha$ و $\beta$ دارد. معادله $x^2 + (\alpha + \beta)x + \alpha\beta = 0$ ریشه‌های $\gamma$ و $\delta$ دارد. $\gamma + \delta$ و $\gamma\delta$ را بر حسب $b$ و $c$ بیان کنید.
\item فرض کنید $\alpha, \beta$ ریشه‌های $x^2 + mx + n = 0$ باشند. اگر $\alpha^2 + \beta^2 = 6$ و $\alpha^3 + \beta^3 = 14$، $m$ و $n$ را بیابید.
\item فرض کنید $p, q$ ریشه‌های $x^2 + x + 1 = 0$ باشند. $p^3 + q^3$ و $p^6 + q^6$ را محاسبه کنید. توضیح دهید چرا دومی برابر ۲ است.
\item اگر $\alpha, \beta$ ریشه‌های $x^2 + 2ax + b = 0$ و $\gamma, \delta$ ریشه‌های $x^2 + 2ax + c = 0$ باشند (با $a^2 > b > c$)، شرط روی $a, b, c$ را بیابید تا چهار ریشه به ترتیب مرتب شده تشکیل دنباله حسابی بدهند.
\item اگر $\alpha, \beta$ ریشه‌های $x^2 + bx + c = 0$ و $\beta, \gamma$ ریشه‌های $x^2 + cx + b = 0$ با $\alpha \neq \beta$ باشند، $\alpha + \gamma$ را بر حسب $b$ و $c$ بیابید.
\item با دانستن اینکه $\alpha, \beta$ ریشه‌های $x^2 + x + 1 = 0$ هستند، معادله‌ای بیابید که ریشه‌های آن $\alpha^{2024}$ و $\beta^{2024}$ باشند.
\item فرض کنید $\alpha, \beta$ ریشه‌های $x^2 + (k^2 - 4)x + k = 0$ باشند. برای چه مقادیر $k$ ریشه‌ها مختلط (غیرحقیقی) هستند؟
\item معادله $x^2 + bx + c = 0$ ریشه‌های $\alpha, \beta$ دارد. با دانستن $|\alpha| + |\beta| = 5$ و $\alpha\beta = -6$، $b$ را بیابید.
\item فرض کنید $\alpha, \beta$ ریشه‌های $x^2 + x - 3 = 0$ باشند. $\alpha^7 + \beta^7$ را محاسبه کنید.
\end{enumerate}
